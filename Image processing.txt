import cv2
import numpy as np

def inspect_gear(ideal_path, sample_path):
    ideal_img = cv2.imread(ideal_path, cv2.IMREAD_GRAYSCALE)
    sample_img = cv2.imread(sample_path, cv2.IMREAD_GRAYSCALE)

    
    if ideal_img is None or sample_img is None:
        print(f"Error: Could not load images. Check paths.")
        return

    # Thresholding
    _, ideal_thresh = cv2.threshold(ideal_img, 128, 255, cv2.THRESH_BINARY_INV)
    _, sample_thresh = cv2.threshold(sample_img, 128, 255, cv2.THRESH_BINARY_INV)

   
    diff_img = cv2.bitwise_xor(ideal_thresh, sample_thresh)

   
    contours, _ = cv2.findContours(diff_img, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

  
    num_defects = len(contours)
    if num_defects == 0:
        print(f"{sample_path}: No defects found (Normal gear).")
    else:
        
        print(f"{sample_path}: Detected {num_defects} potential defects.")
